# COMO PROBAR TODOS LOS ENDPOINTS

Servidor: http://localhost:8001

## USUARIOS

Obtener todos los usuarios
GET /users/getAll/

Obtener usuario por ID
GET /users/getById/1

Crear usuario
POST /users/createUser/
{
  "name": "Pedro",
  "lastName": "García",
  "password": "mypassword123" 
}

Validar usuario
POST /users/validateUser/
{
  "name": "Juan",
  "password": "123456"
}

Eliminar usuario
DELETE /users/deleteUser/4

## PRODUCTOS

Obtener todos los productos
GET /products/getAllProducts

Obtener producto por ID
GET /products/getProduct/1

Crear producto
POST /products/createProducts
{
  "name": "Samsung Galaxy S23",
  "price": 899.99,
  "quantity": 15,
  "status": "active",
  "categoryId": 1
}

Actualizar producto
PUT /products/updateProducts/1
{
  "name": "Laptop HP Pavilion",
  "price": 1199.99,
  "quantity": 12,
  "status": "active"
}

Eliminar producto
DELETE /products/deleteProduct/8

## PEDIDOS

Crear pedido
POST /orders/
{
  "userId": 1,
  "total": 1329.98,
  "orderDetails": [
    {
      "productId": 1,
      "quantity": 1,
      "price": 999.99
    },
    {
      "productId": 2,
      "quantity": 1,
      "price": 329.99
    }
  ]
}

Obtener todos los pedidos
GET /orders/getOrder

Obtener pedido por ID
GET /orders/getOrderById/1

Obtener detalle de pedido
GET /orders/getDetailOrderById/1

Eliminar pedido
DELETE /orders/deleteOrder/1
